\documentclass[border = 2mm]{standalone}
\usepackage[utf8]{inputenc}
\usepackage{tikz}
\input{multimeter.tex}

% Voltage of the panel without load: the plug of the panel is taken out
% of the charger, and the multimeter, as a voltmeter, reads it alone.

\begin{document}
\begin{tikzpicture}

\node[module=gray!15] (panel) at (0,2.5) {\textbf{Solar panel}\\
    \scriptsize in the sun};
\draw[wirePanel] (panel.south) -- (0,0.45);
\dcPlug{plug}{0,0}
\node[note, left] at (-0.6,0) {plug of the panel,\\out of the charger:\\
    + inside, $-$ outside};

\multimeter{dmm}{5,0.5}{V$\,\overline{\cdots}$ 20}{V}

% Red probe into the hole of the plug, black probe on its sleeve
\draw[probeRed] (dmm-V.south) -- ++(0,-0.9) -| (0,-1.3) -- (plug-plus);
\draw[probeBlack] (dmm-COM.south) -- ++(0,-0.6) -| (plug-minus);
\node[note] at (5,-2.8) {red lead in V$\Omega$,\\black lead in COM};

\end{tikzpicture}
\end{document}
