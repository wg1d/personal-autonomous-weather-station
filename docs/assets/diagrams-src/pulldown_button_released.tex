\begin{circuitikz}[scale=1.5]

% 3.3V Source with European symbol
\draw[thick] (-0.25, 4) -- (0.25, 4);
\node[above] at (0, 4) {VCC};
\node[right] at (0.25, 4) {+3.3V};

% Released Push button (open) connected to 3.3V
\draw (0, 4)
    to[push button, l=Released Button (open), i=$I_{sw}{=}0\text{ A}$] (0, 2)
    coordinate (junction);

% Pull-Down resistor (released state) connected to GND
\draw (junction)
    to[R, l=$R_{pd}$ ($10\text{ k}\Omega$), i=$I_{pd}{\approx}0\text{ A}$, v=$U_R{\approx}0\text{ V}$] (0, 0)
    node[ground] {};

% Connection to the ESP32 pin
\draw (junction)
    to[short, i_=$I_{in}{\approx}0\text{ A}$, *-o] (2.5, 2) node[right, font=\bfseries\sffamily] {Pin (Logic LOW)};

% Explicit Vin tension
\draw (2.5, 2) to[open, v^=$V_{in}{\approx}0\text{ V}$] (2.5, 0) node[ground] {};
