\begin{circuitikz}

% Place ESP32 Wrover DevKit v2.1
\pic (esp) at (0,0) {upesy_esp32};

% Place BME280 exactly as in the first diagram
\pic (bme) at (8,4) {bme280};

% Place SD Card Reader below the BME280
\pic (sd) at (8,-2) {sdcard};

% Wires

% --- BME280 Wires (Unchanged from original) ---
% GND: ESP32 right side -> BME280 GND
\draw[black, thick] (esp-GND) -- ++(1.0,0) |- (bme-GND);

% 3V3: ESP32 right side -> BME280 VIN
\draw[red, thick] (esp-3V3) -- ++(1.5,0) |- (bme-VIN);

% SDA: ESP32 Pin 21 right side -> BME280 SDA
\draw[blue, thick] (esp-21) -- ++(0.5,0) |- (bme-SDA);

% SCL: ESP32 Pin 22 right side -> BME280 SCL
\draw[green!70!black, thick] (esp-22) -- ++(1.2,0) |- (bme-SCL);


% --- SD Card Wires ---
% VCC: ESP32 Pin 5V left side -> SD Card 5V (Route under)
\draw[red, thick] (esp-5V) -- ++(-0.5,0) |- (-0.5, -1.8) -- (7.8, -1.8) |- (sd-5V);

% GND: ESP32 right side -> SD Card GND (Branch off existing GND wire)
\draw[black, thick] (esp-GND) -- ++(1.8,0) |- (sd-GND);

% MISO: ESP32 Pin 19 right side -> SD Card MISO
\draw[blue, thick] (esp-19) -- ++(0.8,0) |- (sd-MISO);

% MOSI: ESP32 Pin 23 right side -> SD Card MOSI
\draw[green!70!black, thick] (esp-23) -- ++(2.0,0) |- (sd-MOSI);

% SCLK: ESP32 Pin 18 right side -> SD Card SCLK
\draw[orange, thick] (esp-18) -- ++(2.2,0) |- (sd-SCLK);

% CS: ESP32 Pin 5 right side -> SD Card CS (Route slightly down to avoid BME SDA overlap)
\draw[purple, thick] (esp-5) -- (5.2, 4.5) -- (5.2, 4.2) -- (7.4, 4.2) |- (sd-CS);



\end{circuitikz}
