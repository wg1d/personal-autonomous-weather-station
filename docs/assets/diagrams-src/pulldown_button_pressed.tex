\begin{circuitikz}[scale=1.5]

\draw (0, 4)
    to[short, l_=Pressed Button (closed), i=$I_{sw}{=}0.33\text{ mA}$] (0, 2)
    coordinate (junction);

% Pull-Down resistor (pressed state) connected to GND
\draw (junction)
    to[R, l=$R_{pd}$ ($10\text{ k}\Omega$), i=$I_{pd}{=}0.33\text{ mA}$, v=$U_R{=}3.3\text{ V}$] (0, 0)
    node[ground] {};

% Connection to the ESP32 pin
\draw (junction)
    to[short, i_=$I_{in}{\approx}0\text{ A}$, *-o] (2.5, 2) node[right, font=\bfseries\sffamily] {Pin (Logic HIGH)};

% Explicit Vin tension
\draw (2.5, 2) to[open, v^=$V_{in}{\approx}3.3\text{ V}$] (2.5, 0) node[ground] {};
